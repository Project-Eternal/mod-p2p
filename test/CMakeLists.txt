# Native tests of the portable core of mod-p2p (src/core): cmake -S test -B $TMPDIR/p2p-test && cmake --build $TMPDIR/p2p-test && ctest --test-dir $TMPDIR/p2p-test
cmake_minimum_required(VERSION 3.16)
project(p2p_test CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL 3 REQUIRED)
file(GLOB CORE ${CMAKE_CURRENT_SOURCE_DIR}/../src/core/*.cpp)
add_library(p2pcore STATIC ${CORE})
target_link_libraries(p2pcore PUBLIC OpenSSL::Crypto)
target_compile_options(p2pcore PUBLIC -Wall -Wextra -Wno-unused-parameter)
enable_testing()
file(GLOB TESTS ${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp)
foreach(t ${TESTS})
  get_filename_component(n ${t} NAME_WE)
  add_executable(${n} ${t})
  target_link_libraries(${n} p2pcore)
  add_test(NAME ${n} COMMAND ${n})
endforeach()
